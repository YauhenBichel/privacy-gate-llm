#     --index-url https://pypi.org/simple --extra-index-url https://download.pytorch.org/whl/cpu \
#     --index-strategy unsafe-best-match --no-emit-index-url --no-header --no-annotate \
#     | sed 's/+cpu$//'
# and replace the pins below this header with its output. aarch64-manylinux_2_28 resolves to the same pins.
